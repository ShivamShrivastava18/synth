```python
"""Cloud SQL Postgres accessor."""
import os
import pandas as pd
from sqlalchemy import create_engine, text

def sample_rows(table: str, n: int = 5000, *, db: str = "prod") -> pd.DataFrame:
    dsn = os.environ[f"{db.upper()}_DB_DSN"]
    engine = create_engine(dsn)
    with engine.connect() as conn:
        return pd.read_sql(
            text(f"SELECT * FROM {table} ORDER BY random() LIMIT :n"),
            conn,
            params={"n": n},
        )

def describe_schema(table: str, *, db: str = "prod") -> dict:
    dsn = os.environ[f"{db.upper()}_DB_DSN"]
    engine = create_engine(dsn)
    with engine.connect() as conn:
        rows = conn.execute(text("""
            SELECT column_name, data_type
            FROM information_schema.columns
            WHERE table_name = :t
            ORDER BY ordinal_position
        """), {"t": table}).fetchall()
    return {"table": table, "columns": [{"name": r[0], "type": r[1]} for r in rows]}

def write_rows(table: str, df: pd.DataFrame, *, db: str = "staging") -> int:
    dsn = os.environ[f"{db.upper()}_DB_DSN"]
    engine = create_engine(dsn)
    df.to_sql(table, engine, if_exists="replace", index=False, chunksize=5_000)
    return len(df)
```

```python
"""Distance to Closest Record privacy metric.

For each synthetic row, compute Euclidean distance to nearest real row
(in scaled feature space), normalize by median real-to-real distance.
Returns the minimum ratio across all synthetic rows.
"""
import numpy as np
import pandas as pd
from sklearn.neighbors import NearestNeighbors
from sklearn.preprocessing import StandardScaler

def compute_dcr_min(synth: pd.DataFrame, real: pd.DataFrame, sample_for_median: int = 1000) -> float:
    real = real.select_dtypes(include="number").dropna()
    synth = synth.select_dtypes(include="number").dropna()
    cols = [c for c in real.columns if c in synth.columns]
    real_v = real[cols].to_numpy()
    synth_v = synth[cols].to_numpy()
    if len(real_v) == 0 or len(synth_v) == 0:
        return 0.0
    scaler = StandardScaler().fit(real_v)
    real_s = scaler.transform(real_v)
    synth_s = scaler.transform(synth_v)
    nn = NearestNeighbors(n_neighbors=1).fit(real_s)
    synth_to_real, _ = nn.kneighbors(synth_s)
    # Median pairwise distance among a real sample
    n = min(sample_for_median, len(real_s))
    idx = np.random.default_rng(0).choice(len(real_s), n, replace=False)
    sample = real_s[idx]
    nn2 = NearestNeighbors(n_neighbors=2).fit(sample)
    real_to_real, _ = nn2.kneighbors(sample)
    median_real = np.median(real_to_real[:, 1])
    if median_real == 0:
        return 0.0
    return float(np.min(synth_to_real) / median_real)
```

```python
"""Train Synthetic, Test Real (TSTR). Trains XGBoost on synthetic, evaluates AUC on real holdout."""
import numpy as np
import pandas as pd
from sklearn.metrics import roc_auc_score
from sklearn.model_selection import train_test_split
import xgboost as xgb

def compute_tstr_auc(synth: pd.DataFrame, real: pd.DataFrame, target_col: str) -> float:
    if target_col not in real.columns:
        return float("nan")
    real_train, real_test = train_test_split(real, test_size=0.3, random_state=42)
    # Use real_test only for evaluation
    X_synth = synth.drop(columns=[target_col]).select_dtypes(include="number").dropna()
    y_synth = synth.loc[X_synth.index, target_col]
    X_real_test = real_test.drop(columns=[target_col]).select_dtypes(include="number")
    X_real_test = X_real_test[X_synth.columns]
    y_real_test = real_test.loc[X_real_test.index, target_col]
    clf = xgb.XGBClassifier(n_estimators=50, max_depth=4, use_label_encoder=False, eval_metric="logloss")
    clf.fit(X_synth, y_synth)
    proba = clf.predict_proba(X_real_test)[:, 1]
    return float(roc_auc_score(y_real_test, proba))
```

- [ ] **Step 4: Implement KS / JS**

Write `synth/engine/src/synth_engine/validators/ks_js.py`:
```python
"""Kolmogorov-Smirnov (numeric) and Jensen-Shannon (categorical) distributional similarity."""
import numpy as np
import pandas as pd
from scipy.stats import ks_2samp
from scipy.spatial.distance import jensenshannon

def compute_ks_avg(synth: pd.DataFrame, real: pd.DataFrame) -> float:
    num_cols = real.select_dtypes(include="number").columns
    num_cols = [c for c in num_cols if c in synth.columns]
    if not num_cols:
        return 0.0
    stats = [ks_2samp(real[c].dropna(), synth[c].dropna()).statistic for c in num_cols]
    return float(np.mean(stats))

def compute_js_avg(synth: pd.DataFrame, real: pd.DataFrame) -> float:
    cat_cols = real.select_dtypes(exclude="number").columns
    cat_cols = [c for c in cat_cols if c in synth.columns]
    if not cat_cols:
        return 0.0
    out = []
    for c in cat_cols:
        cats = sorted(set(real[c].dropna()) | set(synth[c].dropna()))
        p = real[c].value_counts(normalize=True).reindex(cats, fill_value=1e-9).to_numpy()
        q = synth[c].value_counts(normalize=True).reindex(cats, fill_value=1e-9).to_numpy()
        out.append(jensenshannon(p, q))
    return float(np.mean(out))
```

```python
"""FastAPI app. Stateless. Reads from Cloud SQL via data.py."""
import os
import uuid
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from google.cloud import storage
import pandas as pd

from .data import sample_rows, describe_schema, write_rows
from .generators.gaussian_copula import generate as gc_generate
from .generators.conditional_histogram import generate as ch_generate
from .validators.tstr import compute_tstr_auc
from .validators.ks_js import compute_ks_avg, compute_js_avg
from .validators.dcr import compute_dcr_min

app = FastAPI(title="Synth Engine")
GENERATORS = {"gaussian_copula": gc_generate, "conditional_histogram": ch_generate}

class GenerateReq(BaseModel):
    source_table: str
    engine: str = "gaussian_copula"
    n: int = 50_000
    run_id: str | None = None

class ValidateReq(BaseModel):
    run_id: str
    source_table: str
    target_col: str | None = None

class PushReq(BaseModel):
    run_id: str
    destination_table: str

# In-memory cache of synthetic outputs by run_id, plus GCS for persistence.
_SYNTH_CACHE: dict[str, pd.DataFrame] = {}

@app.get("/health")
def health():
    return {"ok": True}

@app.get("/schema/{table}")
def schema(table: str):
    return describe_schema(table)

@app.post("/generate")
def generate(req: GenerateReq):
    if req.engine not in GENERATORS:
        raise HTTPException(400, f"unknown engine: {req.engine}")
    try:
        real = sample_rows(req.source_table, n=20_000)
    except Exception as e:
        raise HTTPException(503, f"source db unavailable: {e}")
    synth = GENERATORS[req.engine](real, n=req.n)
    run_id = req.run_id or str(uuid.uuid4())
    _SYNTH_CACHE[run_id] = synth
    return {"run_id": run_id, "rows": len(synth), "engine": req.engine, "columns": list(synth.columns)}

@app.post("/validate")
def validate(req: ValidateReq):
    synth = _SYNTH_CACHE.get(req.run_id)
    if synth is None:
        raise HTTPException(404, "run not found")
    real = sample_rows(req.source_table, n=10_000)
    metrics = {
        "TSTR": compute_tstr_auc(synth, real, target_col=req.target_col) if req.target_col else None,
        "KS_avg": compute_ks_avg(synth, real),
        "JS_avg": compute_js_avg(synth, real),
        "DCR_min": compute_dcr_min(synth, real),
    }
    return metrics

@app.post("/push")
def push(req: PushReq):
    synth = _SYNTH_CACHE.get(req.run_id)
    if synth is None:
        raise HTTPException(404, "run not found")
    n = write_rows(req.destination_table, synth, db="staging")
    return {"pushed": n}

@app.get("/runs/{run_id}/plot_sample")
def plot_sample(run_id: str, source_table: str, columns: str):
    synth = _SYNTH_CACHE.get(run_id)
    if synth is None:
        raise HTTPException(404, "run not found")
    col_list = columns.split(",")
    real = sample_rows(source_table, n=500)
    return {
        "real": real[col_list].head(500).to_dict(orient="records"),
        "synth": synth[col_list].head(500).to_dict(orient="records"),
    }
```

```python
"""Agent tool: discover schema of a source table via Fivetran MCP.

Fallback: if Fivetran cannot return schema for the configured source,
call Engine API /schema/{table} directly.
"""
import os
import httpx

ENGINE_URL = os.environ.get("ENGINE_URL", "")
FIVETRAN_MCP_URL = os.environ.get("FIVETRAN_MCP_URL", "")
FIVETRAN_API_KEY = os.environ.get("FIVETRAN_API_KEY", "")
FIVETRAN_API_SECRET = os.environ.get("FIVETRAN_API_SECRET", "")

def discover_schema(source_table: str) -> dict:
    if FIVETRAN_MCP_URL:
        try:
            resp = httpx.post(
                f"{FIVETRAN_MCP_URL}/tools/get_schema",
                auth=(FIVETRAN_API_KEY, FIVETRAN_API_SECRET),
                json={"table": source_table},
                timeout=30,
            )
            if resp.status_code == 200:
                return resp.json()
        except Exception:
            pass
    # Fallback to engine
    resp = httpx.get(f"{ENGINE_URL}/schema/{source_table}", timeout=30)
    resp.raise_for_status()
    return resp.json()
```

```python
"""Agent tool: poll Firestore until verdict, max 30 minutes."""
import time
from google.cloud import firestore

db = firestore.Client()

def request_human_approval(run_id: str, timeout_seconds: int = 1800) -> dict:
    start = time.time()
    while time.time() - start < timeout_seconds:
        doc = db.collection("runs").document(run_id).get()
        if doc.exists:
            data = doc.to_dict()
            verdict = data.get("approval_verdict")
            if verdict in ("approved", "rejected"):
                return {"verdict": verdict}
        time.sleep(5)
    return {"verdict": "timeout"}
```

## Phase 5 — Fivetran push + branching

**When:** Mon-Thu Jun 1-4 (~12h)

### Task 5.1: Implement `push_destination` tool

Wraps Fivetran MCP push (or fallback direct write).

- [ ] **Step 1**: Write `agent/tools/push_destination.py`:

```python
import os, httpx
ENGINE_URL = os.environ["ENGINE_URL"]
FIVETRAN_MCP_URL = os.environ.get("FIVETRAN_MCP_URL", "")
FIVETRAN_API_KEY = os.environ.get("FIVETRAN_API_KEY", "")
FIVETRAN_API_SECRET = os.environ.get("FIVETRAN_API_SECRET", "")

def push_destination(run_id: str, destination_table: str) -> dict:
    if FIVETRAN_MCP_URL:
        try:
            resp = httpx.post(
                f"{FIVETRAN_MCP_URL}/tools/load_rows",
                auth=(FIVETRAN_API_KEY, FIVETRAN_API_SECRET),
                json={"run_id": run_id, "destination_table": destination_table},
                timeout=300,
            )
            if resp.status_code == 200:
                return resp.json()
        except Exception:
            pass
    # Fallback: direct via engine
    resp = httpx.post(
        f"{ENGINE_URL}/push",
        json={"run_id": run_id, "destination_table": destination_table},
        timeout=300,
    )
    resp.raise_for_status()
    return resp.json()
```
